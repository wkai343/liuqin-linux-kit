cmake_minimum_required(VERSION 3.20)
project(liuqin-obs-v4l2 VERSION 0.1.0 LANGUAGES C)

find_package(PkgConfig REQUIRED)
pkg_check_modules(OBS REQUIRED libobs>=32)
pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec>=63)
pkg_check_modules(AVUTIL REQUIRED IMPORTED_TARGET libavutil>=61)
# Fedora's libobs.pc names the parent include directory.
find_path(OBS_INCLUDE_DIR obs-module.h HINTS ${OBS_INCLUDE_DIRS} PATH_SUFFIXES obs REQUIRED)
find_library(OBS_LIBRARY obs HINTS ${OBS_LIBRARY_DIRS} REQUIRED)

add_library(liuqin-v4l2 MODULE liuqin-v4l2.c)
target_compile_features(liuqin-v4l2 PRIVATE c_std_11)
target_compile_options(liuqin-v4l2 PRIVATE -Wall -Wextra -Werror ${OBS_CFLAGS_OTHER})
target_include_directories(liuqin-v4l2 PRIVATE ${OBS_INCLUDE_DIR} ${OBS_INCLUDE_DIRS})
target_link_libraries(liuqin-v4l2 PRIVATE ${OBS_LIBRARY} PkgConfig::AVCODEC PkgConfig::AVUTIL)
set_target_properties(liuqin-v4l2 PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
# Both layouts add a new module; neither replaces an official OBS file.
option(OBS_INSTALL_SYSTEM "Use Fedora's system OBS plugin search paths" OFF)
set(OBS_PLUGIN_DESTINATION "liuqin-v4l2" CACHE STRING "Relative OBS plugin directory")
if(OBS_INSTALL_SYSTEM)
  set(OBS_MODULE_BIN "lib64/obs-plugins")
  set(OBS_MODULE_DATA "share/obs/obs-plugins/liuqin-v4l2")
else()
  set(OBS_MODULE_BIN "${OBS_PLUGIN_DESTINATION}/bin/64bit")
  set(OBS_MODULE_DATA "${OBS_PLUGIN_DESTINATION}/data")
endif()
install(TARGETS liuqin-v4l2 LIBRARY DESTINATION "${OBS_MODULE_BIN}")
install(DIRECTORY data/ DESTINATION "${OBS_MODULE_DATA}")
install(FILES README.md COPYING DESTINATION "${OBS_MODULE_DATA}")

option(BUILD_TESTING "Build the packet adapter regression" OFF)
if(BUILD_TESTING)
  enable_testing()
  add_executable(obs-v4l2-packets ../../tests/obs-v4l2-packets.c)
  target_compile_features(obs-v4l2-packets PRIVATE c_std_11)
  target_compile_options(obs-v4l2-packets PRIVATE -Wall -Wextra -Werror -UNDEBUG ${OBS_CFLAGS_OTHER})
  target_include_directories(obs-v4l2-packets PRIVATE ${OBS_INCLUDE_DIR} ${OBS_INCLUDE_DIRS})
  target_link_libraries(obs-v4l2-packets PRIVATE ${OBS_LIBRARY} PkgConfig::AVCODEC PkgConfig::AVUTIL)
  add_test(NAME obs-v4l2-packets COMMAND obs-v4l2-packets)
endif()
