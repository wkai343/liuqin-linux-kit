cmake_minimum_required(VERSION 3.16)
project(liuqin_recovery_ui C CXX)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(LV_USE_FIND_PACKAGE OFF CACHE BOOL "" FORCE)
set(LV_USE_PKG_CONFIG OFF CACHE BOOL "" FORCE)
find_package(Threads REQUIRED)
add_subdirectory(${LVGL_SOURCE} ${CMAKE_BINARY_DIR}/lvgl)
add_executable(liuqin-recovery-ui main.c platform.c touch_input.c ${CJSON_SOURCE}/cJSON.c ${FONT_SOURCE}/ui_font_20.c ${FONT_SOURCE}/ui_font_26.c ${FONT_SOURCE}/ui_font_34.c)
target_include_directories(liuqin-recovery-ui PRIVATE ${CJSON_SOURCE})
target_link_libraries(liuqin-recovery-ui PRIVATE lvgl m Threads::Threads)
target_compile_options(liuqin-recovery-ui PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
if(STATIC_UI)
 target_compile_options(liuqin-recovery-ui PRIVATE -Os -ffunction-sections -fdata-sections)
 target_link_options(liuqin-recovery-ui PRIVATE -static -s -Wl,--gc-sections)
endif()
option(UI_EVENT_TESTS "Build hardware-free UI event tests" OFF)
if(UI_EVENT_TESTS)
 add_executable(native-touch-input ../../../tests/native-touch-input.c touch_input.c)
 target_include_directories(native-touch-input PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 add_executable(native-ui-events ../../../tests/native-ui-events.c platform.c touch_input.c ${CJSON_SOURCE}/cJSON.c ${FONT_SOURCE}/ui_font_20.c ${FONT_SOURCE}/ui_font_26.c ${FONT_SOURCE}/ui_font_34.c)
 target_include_directories(native-ui-events PRIVATE ${CJSON_SOURCE} ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(native-ui-events PRIVATE lvgl m Threads::Threads)
endif()
